from __future__ import annotations

import argparse

from board_manager import BoardManager
from chess_engine import StockfishAnalyzer, find_stockfish_executable
from gui import ChessAssistantApp


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="AI Chess Assistant")
    parser.add_argument("--fen", type=str, default=None, help="Load a custom starting FEN")
    parser.add_argument("--depth", type=int, default=14, help="Engine search depth")
    parser.add_argument("--analysis-time", type=float, default=1.5, help="Engine analysis time in seconds")
    parser.add_argument("--headless", action="store_true", help="Run terminal-only mode without the GUI")
    return parser.parse_args()


def main() -> int:
    args = parse_args()
    board_manager = BoardManager(args.fen)

    if args.headless:
        engine = StockfishAnalyzer(depth=args.depth, analysis_time=args.analysis_time)
        recommendation = engine.recommend_move(board_manager.board)
        if recommendation is None:
            print("Stockfish could not be started or no legal moves are available.")
            return 1

        print(f"Board FEN: {board_manager.board.fen()}")
        print(f"Best move: {recommendation['san']} / {recommendation['uci']}")
        print(f"Evaluation: {recommendation['evaluation']}")
        print(f"Principal variation: {' '.join(recommendation['principal_variation'])}")
        return 0

    engine_path = find_stockfish_executable()
    if engine_path is None:
        print("Stockfish was not found. Install Stockfish or set STOCKFISH_EXECUTABLE.")
        print("GUI will still open, but engine features will be disabled until the binary is available.")

    app = ChessAssistantApp(
        board_manager=board_manager,
        engine_path=engine_path,
        analysis_time=args.analysis_time,
        depth=args.depth,
    )
    app.mainloop()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
